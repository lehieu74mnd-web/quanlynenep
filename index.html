/** * GOOGLE APPS SCRIPT - ĐỒNG BỘ DỮ LIỆU CHUẨN XÁC TUYỆT ĐỐI */ function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const selectedWeek = (e && e.parameter && e.parameter.week) ? Number(e.parameter.week) : 1; let sheetDash = ss.getSheetByName("DASHBOARD_TUAN") || ss.getSheetByName("TONG_KET_35_TUAN"); const scoresMap = {}; if (sheetDash) { // 1. Tự động ghi số tuần vào ô I2 của sheet Dashboard để kích hoạt công thức tính lại try { sheetDash.getRange("I2").setValue(selectedWeek); SpreadsheetApp.flush(); } catch (err) {} // 2. Quét dữ liệu từ dòng 6 đến hết danh sách 30 lớp const lastRow = sheetDash.getLastRow(); if (lastRow >= 6) { const dataDash = sheetDash.getRange(6, 1, lastRow - 5, 12).getValues(); for (let r = 0; r < dataDash.length; r++) { const row = dataDash[r]; const lop = row[2] ? row[2].toString().trim() : ""; // Cột C (index 2): Tên Lớp if (lop && (lop.startsWith("10") || lop.startsWith("11") || lop.startsWith("12"))) { scoresMap[lop] = { coDoMinus: Math.abs(Number(row[4]) || 0), // Cột E (index 4): Trừ cờ đỏ sdbMinus: Math.abs(Number(row[5]) || 0), // Cột F (index 5): Trừ SĐB vangMinus: Math.abs(Number(row[6]) || 0), // Cột G (index 6): Trừ vắng rewardPlus: Math.abs(Number(row[7]) || 0), // Cột H (index 7): Điểm thưởng totalScore: Number(row[8]) || 320, // Cột I (index 8): Tổng điểm avgScore: Number(row[9]) || 10, // Cột J (index 9): Điểm TB rankInKhoi: Number(row[10]) || 1 // Cột K (index 10): Hạng khối }; } } } } // 3. Đọc dữ liệu chi tiết sự việc từ NHAT_KY_VI_PHAM cho Tab 2 let sheetNK = ss.getSheetByName("NHAT_KY_VI_PHAM"); const incidents = []; if (sheetNK && sheetNK.getLastRow() > 1) { const dataNK = sheetNK.getDataRange().getValues(); for (let r = 1; r < dataNK.length; r++) { const row = dataNK[r]; const w = Number(row[1]); const targetW = isNaN(w) || w === 0 ? 1 : w; if (targetW === selectedWeek && (row[0] || row[5])) { const rawMinus = Number(row[9]) || 0; const rawPlus = Number(row[10]) || 0; let finalPoints = rawPlus > 0 ? Math.abs(rawPlus) : (rawMinus !== 0 ? -Math.abs(rawMinus) : 0); let dateStr = ""; const rawDate = row[2]; if (rawDate) { if (rawDate instanceof Date) { dateStr = Utilities.formatDate(rawDate, "GMT+7", "dd/MM/yyyy"); } else { const parsed = new Date(rawDate); dateStr = !isNaN(parsed.getTime()) ? Utilities.formatDate(parsed, "GMT+7", "dd/MM/yyyy") : rawDate.toString(); } } incidents.push({ id: row[0], week: targetW, date: dateStr, day: row[3] || "", session: row[4] || "", lop: row[5] ? row[5].toString().trim() : "", student: row[6] || "", source: row[7] || "", category: row[8] || "", points: finalPoints, note: row[11] || "" }); } } } return ContentService.createTextOutput(JSON.stringify({ status: "success", week: selectedWeek, scoresMap: scoresMap, incidents: incidents })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let postData = {}; try { postData = JSON.parse(e.postData.contents); } catch (err) { postData = {}; } const action = postData.action || ""; if (action === "updateAttendance" || action === "saveAttendance" || action === "addAttendance" || postData.attendance || postData.data || postData.absents) { let sheetCC = ss.getSheetByName("CHUYEN_CAN_35W") || ss.getSheetByName("CHUYEN_CAN_32W"); if (!sheetCC) { sheetCC = ss.insertSheet("CHUYEN_CAN_35W"); sheetCC.appendRow(["STT", "KHÔI", "LỚP"]); } const week = Number(postData.week) || 1; const targetCol = week + 3; const attendanceData = postData.attendance || postData.data || postData.absents || postData; const dataCC = sheetCC.getDataRange().getValues(); for (let r = 5; r < dataCC.length; r++) { const lop = dataCC[r][2] ? dataCC[r][2].toString().trim() : ""; if (lop && attendanceData[lop] !== undefined) { const item = attendanceData[lop]; let totalMinus = 0; if (typeof item === 'object' && item !== null) { const p = Number(item.p) || Number(item.coPhép) || 0; const kp = Number(item.kp) || Number(item.khongPhép) || 0; totalMinus = (p * 1) + (kp * 10); } else { totalMinus = Number(item) || 0; } sheetCC.getRange(r + 1, targetCol).setValue(totalMinus); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Đã cập nhật điểm chuyên cần thành công!" })).setMimeType(ContentService.MimeType.JSON); } if (action === "deleteIncident") { let sheetNK = ss.getSheetByName("NHAT_KY_VI_PHAM"); if (sheetNK && sheetNK.getLastRow() > 1) { const targetId = String(postData.id || "").trim(); const dataNK = sheetNK.getDataRange().getValues(); for (let r = dataNK.length - 1; r >= 1; r--) { const rowId = dataNK[r][0] !== undefined && dataNK[r][0] !== null ? dataNK[r][0].toString().trim() : ""; if (rowId === targetId || (targetId !== "" && rowId.includes(targetId))) { sheetNK.deleteRow(r + 1); break; } } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } if (action === "addIncident") { let sheetNK = ss.getSheetByName("NHAT_KY_VI_PHAM"); if (!sheetNK) { sheetNK = ss.insertSheet("NHAT_KY_VI_PHAM"); sheetNK.appendRow(["ID", "TUẦN", "NGÀY", "THỨ", "BUỔI", "LỚP", "HỌC SINH", "NGUỒN", "NỘI QUY", "TRỪ", "CỘNG", "GHI CHÚ"]); } const inc = postData.incident || {}; const targetWeek = Number(postData.week) || Number(inc.week) || 1; const pts = Number(postData.incident.points) || Number(inc.points) || 0; const minus = pts < 0 ? Math.abs(pts) : 0; const plus = pts > 0 ? Math.abs(pts) : 0; sheetNK.appendRow([inc.id || Date.now(), targetWeek, inc.date || "", inc.day || "", inc.session || "", inc.lop || "", inc.student || "", inc.source || "", inc.category || "", minus, plus, inc.note || ""]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Không rõ hành động" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
